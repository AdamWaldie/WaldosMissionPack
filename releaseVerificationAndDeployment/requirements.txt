# Python dependencies for the WaldosMissionPack dev/CI tooling in this directory.
#
# sqf_validator.py and config_style_checker.py use only the Python standard
# library, so no install is required to run them.
#
# sqf_linter.py is an optional local linter that depends on the third-party
# "sqf" module (LordGolias/sqf), published on PyPI as `sqflint`. Install with:
#   pip install -r releaseVerificationAndDeployment/requirements.txt
#
# Note: CI runs the linter via the arma-actions/sqflint GitHub Action
# (see .github/workflows/testing.yml), not this file. This requirements file
# exists so the local linter is reproducible for contributors.
sqflint

# Pillow is required by generateLoadingScreen.py, which renders the cover image
# (Pictures/loading.jpg). It is installed by the update-cover.yml and deploy.yml
# workflows and by deploy.sh; install it locally to regenerate the cover by hand.
Pillow

# PyYAML is required by claude_skill_validator.py and package_claude_skill_upload.py
# to parse SKILL.md's frontmatter the same way claude.ai's own uploader does. It is
# installed by the testing.yml and deploy.yml workflows; install it locally to run
# either script or regenerate the claude.ai-upload-ready skill package by hand.
PyYAML
